\newglossaryentry{akquisition}{%
    name={Akquisition},%
    description={%
        Erwerb neuer Kunstobjekte für eine Sammlung, sei es durch Kauf oder
        Schenkung (engl. \textit{acquisition}); im Modell als Detail-, Sub- oder Teilprozess
        \emph{Akquisition durchführen} (engl. \textit{Perform artwork
        acquisition}) abgebildet%
    },%
    sort=akquisition%
}

\newglossaryentry{artothek}{%
    name={Artothek},%
    description={%
        Einrichtung, die Kunstwerke sammelt, verwahrt und verleiht
        (Kunstverleih); im Kontext dieser Arbeit der Ort der Verwahrung und
        Bereitstellung des Kunstbestands%
    },%
    sort=artothek%
}

\newglossaryentry{artworkmanagement}{%
    name={Artwork Management},%
    description={%
        Übergeordneter Prozess-Cluster (Ebene 0) der Bayer AG zur Steuerung
        des gesamten Umgangs mit dem Kunstbestand; umfasst u.\,a. den
        Kernprozess \emph{Collection Management}%
    },%
    sort=artworkmanagement%
}

\newglossaryentry{callactivity}{%
    name={Aufruf-Aktivität (engl. \textit{Call Activity})},%
    description={%
        Aufrufbare Aktivität bzw. aufrufbarer Prozess in der BPMN; verweist
        auf einen eigenständig modellierten, wiederverwendbaren (Sub-)Prozess
        und ermöglicht die modulare Zerlegung des Modells%
    },%
    sort=callactivity%
}

\newglossaryentry{collectionmanagement}{%
    name={Collection Management},%
    description={%
        Englische Bezeichnung der Sammlungsverwaltung; in dieser Arbeit der
        betrachtete Kernprozess des Prozess-Clusters \emph{Artwork Management}%
    },%
    sort=collectionmanagement%
}

\newglossaryentry{culturalaffairs}{%
    name={Cultural Affairs},%
    description={%
        Organisatorischer Bereich innerhalb der Unternehmensfunktion
        \emph{Communications and Public Affairs} der Bayer AG, der die
        Kunstsammlung verantwortet%
    },%
    sort=culturalaffairs%
}

\newglossaryentry{deakzession}{%
    name={Deakzession},%
    description={%
        Planmässiges Ausscheiden von Objekten aus einer Sammlung, etwa durch
        Verkauf oder Abgabe (engl. \textit{deaccession}); bildet das Gegenstück
        zur Akquisition%
    },%
    sort=deakzession%
}

\newglossaryentry{kernprozess}{%
    name={Kernprozess},%
    description={%
        Prozess mit unmittelbarem Beitrag zur Wertschöpfung und Aussenbezug;
        steht im Zentrum der betrieblichen Leistungserstellung%
    },%
    sort=kernprozess%
}

\newglossaryentry{kunstverwalter}{%
    name={Kunstverwalter},%
    description={%
        Für die operative Verwaltung des Kunstbestands zuständige Rolle bzw.
        Stelle; im betrachteten Fall die Tochtergesellschaft Bayer Kultur GmbH%
    },%
    sort=kunstverwalter%
}

\newglossaryentry{kurator}{%
    name={Kurator},%
    description={%
        Fachlich für Auswahl, Betreuung, Erschliessung und wissenschaftliche
        Bearbeitung einer Sammlung verantwortliche Person%
    },%
    sort=kurator%
}

\newglossaryentry{lifescience}{%
    name={Life-Science},%
    description={%
        Wissenschafts- und Branchenfeld an der Schnittstelle von Biologie,
        Medizin, Pharmazie und Agrarwissenschaften; Geschäftsfeld der
        Bayer AG%
    },%
    sort=lifescience%
}

\newglossaryentry{managementprozess}{%
    name={Managementprozess},%
    description={%
        Steuernder und führender Prozess (Planung, Steuerung, Kontrolle), der
        Kern- und Unterstützungsprozesse ausrichtet und lenkt%
    },%
    sort=managementprozess%
}

\newglossaryentry{processengine}{%
    name={Process Engine},%
    description={%
        Softwarekomponente zur Ausführung formal spezifizierter
        Prozessmodelle; setzt ausführbare BPMN-Modelle zur Laufzeit um%
    },%
    sort=processengine%
}

\newglossaryentry{prozesscluster}{%
    name={Prozess-Cluster},%
    description={%
        Thematisch zusammengehörige Gruppe von Prozessen; in dieser Arbeit die
        oberste Modellebene (L0) \emph{Artwork Management}%
    },%
    sort=prozesscluster%
}

\newglossaryentry{publicaffairs}{%
    name={Public Affairs},%
    description={%
        Management der Beziehungen einer Organisation zu politischen und
        gesellschaftlichen Anspruchsgruppen sowie deren Interessenvertretung%
    },%
    sort=publicaffairs%
}

\newglossaryentry{publicrelations}{%
    name={Public Relations},%
    description={%
        Planung und Gestaltung der Öffentlichkeitsarbeit und der
        Kommunikationsbeziehungen einer Organisation zu ihren Anspruchsgruppen%
    },%
    sort=publicrelations%
}

\newglossaryentry{sammlungsverwaltung}{%
    name={Sammlungsverwaltung},%
    description={%
        Gesamtheit der Tätigkeiten zur Erfassung, Pflege, Dokumentation und
        Disposition einer Kunstsammlung (engl. \textit{Collection Management})%
    },%
    sort=sammlungsverwaltung%
}

\newglossaryentry{sphaere}{%
    name={Sphäre},%
    description={%
        In dieser Arbeit verwendete logische Gruppierung thematisch
        zusammengehöriger Prozesse des \emph{Collection Management};
        unterschieden werden die Sphären Akquisition, Deakzession sowie
        Verwaltung und Wartung%
    },%
    sort=sphaere%
}

\newglossaryentry{stakeholder}{%
    name={Stakeholder},%
    description={%
        Person oder Gruppe mit berechtigtem Interesse an einem Prozess oder
        dessen Ergebnissen, etwa Fachbereich, IT oder externe Beteiligte%
    },%
    sort=stakeholder%
}

\newglossaryentry{unterstuetzungsprozess}{%
    name={Unterstützungsprozess},%
    description={%
        Prozess ohne unmittelbaren Wertschöpfungsbeitrag, der die
        Durchführung der Kernprozesse ermöglicht (z.\,B. IT oder
        Verwaltung)%
    },%
    sort=unterstuetzungsprozess%
}

\newglossaryentry{wertschoepfung}{%
    name={Wertschöpfung},%
    description={%
        Schaffung von Wert durch die Transformation von Eingangsgrößen in ein
        Ergebnis höheren Nutzens; zentrales Bezugskriterium zur Abgrenzung von
        Kern- und Unterstützungsprozessen%
    },%
    sort=wertschoepfung%
}

\newglossaryentry{flussobjekte}{%
    name={Flussobjekte},%
    description={%
        Kern-Notationsgruppe der BPMN (engl. \textit{flow objects}); umfasst
        \emph{Ereignisse}, \emph{Aktivitäten} und \emph{Gateways}, die den
        eigentlichen Ablauf eines Prozesses abbilden%
    },%
    sort=flussobjekte%
}

\newglossaryentry{verbindungsobjekte}{%
    name={Verbindungsobjekte},%
    description={%
        Notationsgruppe der BPMN (engl. \textit{connecting objects}) zur
        Verknüpfung von Modellelementen: \emph{Sequenzfluss} (innerhalb eines
        Pools), \emph{Nachrichtenfluss} (zwischen Pools) und \emph{Assoziation}
        (zu Artefakten)%
    },%
    sort=verbindungsobjekte%
}

\newglossaryentry{schwimmbahnen}{%
    name={Schwimmbahnen},%
    description={%
        Strukturierende BPMN-Elemente (engl. \textit{swimlanes}) zur
        organisatorischen Gliederung eines Modells in \emph{Pools} und
        \emph{Bahnen} (Lanes); bilden Zuständigkeiten und beteiligte Akteure (Teilnehmer) ab%
    },%
    sort=schwimmbahnen%
}

\newglossaryentry{artefakte}{%
    name={Artefakte},%
    description={%
        Ergänzende, nicht ablaufsteuernde BPMN-Elemente (engl.
        \textit{artifacts}): \emph{Datenobjekte}, \emph{Gruppen} und
        \emph{Textannotationen}; dienen der Anreicherung, nicht der
        Steuerung des Sequenzflusses%
    },%
    sort=artefakte%
}

\newglossaryentry{prozessstrang}{%
    name={Prozessstrang},%
    description={%
        In dieser Arbeit ein eigenständiger, unabhängig aufrufbarer Teilstrang
        innerhalb des L1-Prozesses \emph{Manage collection}; das Modell
        weist vier solcher Stränge auf, je einen pro Detail- bzw. Subprozess%
    },%
    sort=prozessstrang%
}

\newglossaryentry{deakzessionsliste}{%
    name={Deakzessionsliste},%
    description={%
        Im L2-Prozess \emph{Perform artwork deaczession} (vgl. Kap.~3) erzeugte Liste aller potenziell
        verwertbaren Kunstobjekte, die nicht mehr zu den Sammlungszielen passen
        (engl. \textit{deaccession list}) und führt Status- und
        Vertriebskanal-Attribute zur Steuerung der weiteren Abwicklung%
    },%
    sort=deakzessionsliste%
}

\newglossaryentry{stammdaten}{%
    name={Stammdaten},%
    description={%
        Grundlegende, dauerhaft gültige Beschreibungsdaten eines Objekts
        (engl. \textit{master data}), etwa Bezeichnung, Urheberangaben, Erwerbswert oder
        Inventarnummer; bilden die Basis des Kunstwerkdatensatzes%
    },%
    sort=stammdaten%
}

\newglossaryentry{betriebskosten}{%
    name={Betriebskosten},%
    description={%
        Laufende Kosten für die Erhaltung, Lagerung und Versicherung eines Objekts
        (engl. \textit{run costs}); können etwa durch Schäden und den daraus
        folgenden Restaurierungsbedarf steigen, weshalb eine kontinuierliche
        Pflege mögliche Wertminderungen mindert%
    },%
    sort=betriebskosten%
}

\newglossaryentry{entleiheverwalten}{%
    name={Entleihe der Kunstwerke verwalten},%
    description={%
        Prozess des Clusters \emph{Artwork Management} zur Verwaltung der
        Ausleihe von Kunstwerken aus der Artothek an Mitarbeiter
        (engl. \textit{Manage loan agreement}); regelt Leihvertrag,
        Herausgabe und Rückgabe. In dieser Arbeit nicht modelliert, sondern
        als mögliche Erweiterung im Ausblick genannt%
    },%
    sort=entleiheverwalten%
}
